class HatAnaliziVeGerilimDusumuEkrani extends StatefulWidget {
  final int initialTab;

  const HatAnaliziVeGerilimDusumuEkrani({
    super.key,
    this.initialTab = 0,
  });

  @override
  State<HatAnaliziVeGerilimDusumuEkrani> createState() =>
      _HatAnaliziVeGerilimDusumuEkraniState();
}

class _HatAnaliziVeGerilimDusumuEkraniState
    extends State<HatAnaliziVeGerilimDusumuEkrani> {
  late int sekme;

  @override
  void initState() {
    super.initState();
    sekme = widget.initialTab.clamp(0, 1);
  }

  @override
  Widget build(BuildContext context) {
    return AppScaffold(
      title: 'Hat Analizi',
      info: true,
      onInfo: () => bilgiPopup(
        context,
        'Hat Analizi — Bilgi / Yardım',
        '''Bu araçta önce AG/OG ve Trifaze/Monofaze seçilir. Daha sonra iletken tipi, döşeme şekli, kablo kesiti ve güç faktörü belirlenir.

Hesap yöntemi:
• Trifaze akım: I = P / (√3 × U × cosφ)
• Monofaze akım: I = P / (U × cosφ)
• Trifaze gerilim düşümü: ΔU = √3 × I × (R × cosφ + X × sinφ) × L
• Monofaze gerilim düşümü: ΔU = 2 × I × R × L
• Gerilim düşümü yüzdesi: ΔU% = 100 × ΔU / U

Otomatik seçimde önce iletken tipine ve döşeme şekline uygun kapasite kontrol edilir, ardından izin verilen gerilim düşümü kontrol edilir. Veri tabanında empedans bilgisi bulunmayan bir iletken için gerilim düşümü değeri uydurulmaz.

Referanslar:
• TEDAŞ teknik şartnameleri
• İlgili TS / EN / IEC kablo standartları
• Üretici teknik katalogları
• Dağıtım şirketi ve proje özel şartnameleri

Sonuçlar ön mühendislik kontrolüdür. Kısa devre dayanımı, koruma koordinasyonu, sıcaklık/gruplama düzeltme katsayıları ve saha koşulları ayrıca doğrulanmalıdır.''',
      ),
      body: Column(
        children: [
          Padding(
            padding: const EdgeInsets.fromLTRB(14, 10, 14, 4),
            child: Row(
              children: [
                Expanded(child: _sekmeButton('Hat Analizi', 0)),
                const SizedBox(width: 8),
                Expanded(child: _sekmeButton('Gerilim Düşümü', 1)),
              ],
            ),
          ),
          Expanded(
            child: sekme == 0
                ? const _HatAnaliziTab()
                : const _GerilimDusumuTab(),
          ),
        ],
      ),
    );
  }

  Widget _sekmeButton(String text, int index) {
    final aktif = sekme == index;
    return InkWell(
      borderRadius: BorderRadius.circular(8),
      onTap: () => setState(() => sekme = index),
      child: Container(
        padding: const EdgeInsets.symmetric(vertical: 10),
        decoration: BoxDecoration(
          color: aktif ? cIcon() : cCard(),
          borderRadius: BorderRadius.circular(8),
          border: Border.all(color: cIcon().withValues(alpha: .35)),
        ),
        child: Text(
          text,
          textAlign: TextAlign.center,
          style: TextStyle(
            color: aktif ? Colors.white : cText(),
            fontWeight: FontWeight.w800,
            fontSize: 12,
          ),
        ),
      ),
    );
  }
}

  double? _kapasite(String secim) {
    if (iletkenTipi == 'NYY-Bakır Kablo') {
      final base = fazTipi == 'Monofaze' ? _nyyHava(secim) : _nyyHava(secim);

    }
    if (iletkenTipi == 'NAYY-Aluminyum Kablo') {
      final base = _nayyHava(secim);
      return base == null ? null : base * _paralelSayisi(secim);
    }
    if (iletkenTipi == 'Açık İletken') {
      return _acikIletkenAkimi(secim, worstCase: true);
    }
    // ALPEK için merkezi teknik tablo henüz bağlanmadığı için sayı uydurulmaz.

  }

      // Kullanıcının güncel NAYY tablosunda R/X alanı bu araca bağlanmadı;
      // gerilim düşümü burada tahmin edilmez.
      return null;
    }
    if (iletkenTipi == 'Açık İletken') {
      final r = _acikIletkenR(secim);
      return r == null ? null : r / 1000;
    }
    return null;
  }

  double? _gerilimDusumuYuzde({
    required double current,
    required double distanceM,
    required double voltage,
    required bool threePhase,
    required String secim,
  }) {
    final r = _empedansR(secim);
    if (r == null || r <= 0) return null;

    final x = 0.0;
    final sinPhi = sqrt(max(0.0, 1 - pf * pf));

    final du = threePhase
        ? sqrt(3) *
            current *
            ((r * pf) + (x * sinPhi)) *
            distanceM
        : 2 * current * r * distanceM;

    return du / voltage * 100;
  }

  List<String> _adaylar() {
    final source = _kesitler.where((e) => e.isNotEmpty).toList();
    return source;
  }

  String _otomatikSec(double current, double distanceM) {
    if (iletkenTipi == 'Alpek İletken') {
      return '-';
    }

    for (final candidate in _adaylar()) {
      final cap = _kapasite(candidate);
      if (cap == null || cap < current) continue;

      final du = _gerilimDusumuYuzde(
        current: current,
        distanceM: distanceM,
        voltage: gerilim == 'OG' ? ogVolt : (fazTipi == 'Trifaze' ? agVolt : agMonoVolt),
        threePhase: fazTipi == 'Trifaze',
        secim: candidate,
      );

      // Empedans verisi yoksa otomatik seçimde gerilim düşümü kriteri
      // uydurulmaz; bu durumda yalnızca kapasiteye göre seçim yapılmaz.
      if (du == null) continue;
      if (du <= izinVerilenDusum) return candidate;
    }

    return '-';
  }

  void hesapla() {
    FocusManager.instance.primaryFocus?.unfocus();

    final p = double.tryParse(_gucCtrl.text.replaceAll(',', '.')) ?? 0;
    final l = double.tryParse(_mesafeCtrl.text.replaceAll(',', '.')) ?? 0;

    if (p <= 0 || l <= 0 || pf <= 0) {
      setState(() => hesaplandi = false);
      return;
    }

    final three = fazTipi == 'Trifaze';
    final u = gerilim == 'OG' ? ogVolt : (three ? agVolt : agMonoVolt);
    final i = three
        ? p * 1000 / (sqrt(3) * u * pf)
        : p * 1000 / (u * pf);

    String secim = kabloKesiti;
    if (otomatik) {
      secim = _otomatikSec(i, l);
    }

    if (secim.isEmpty || secim == '-') {
      setState(() {
        akim = i;
        kapasite = 0;
        gerilimDusumu = double.nan;
        gerilimDusumuVolt = double.nan;
        durum = 'UYGUN SEÇİM BULUNAMADI';
        onerilen = '-';
        gerekce = iletkenTipi == 'Alpek İletken'
            ? 'ALPEK için merkezi akım/empedans tablosu bu araca henüz bağlanmadığı için sayısal sonuç uydurulmadı.'
            : 'Seçilen iletken tipi için hem akım taşıma hem de doğrulanabilir gerilim düşümü kriterlerini birlikte sağlayan bir seçim bulunamadı.';
        hesaplandi = true;
      });
      return;
    }

    final cap = _kapasite(secim) ?? 0;
    final du = _gerilimDusumuYuzde(
      current: i,
      distanceM: l,
      voltage: u,
      threePhase: three,
      secim: secim,
    );

    final capOk = cap >= i;
    final duOk = du != null && du <= izinVerilenDusum;
    final uygun = capOk && duOk;

    String text;
    if (uygun) {
      text =
          'Seçilen iletken, verilen güç/mesafe ve referans döşeme koşulunda akım taşıma ve gerilim düşümü açısından ön kontrolden geçiyor.';
    } else {
      final nedenler = <String>[];
      if (!capOk) {
        nedenler.add(
          'Akım ${i.toStringAsFixed(1)} A, referans kapasite ${cap.toStringAsFixed(1)} A',
        );
      }
      if (du == null) {
        nedenler.add('Bu iletken için doğrulanmış R/X verisi bulunmadığından gerilim düşümü hesaplanmadı');
      } else if (!duOk) {
        nedenler.add(
          'gerilim düşümü %${du.toStringAsFixed(2)} ile %$izinVerilenDusum sınırını aşıyor',
        );
      }
      text = nedenler.join('. ');
    }

    final next = _otomatikSec(i, l);

    setState(() {
      akim = i;
      kapasite = cap;
      gerilimDusumu = du ?? double.nan;
      gerilimDusumuVolt = du == null ? double.nan : du / 100 * u;
      durum = uygun ? 'UYGUN' : 'UYGUN DEĞİL';
      onerilen = next;
      onerilenKapasite = next == '-' ? 0 : (_kapasite(next) ?? 0);
      onerilenDusum = next == '-'
          ? double.nan
          : (_gerilimDusumuYuzde(
                current: i,
                distanceM: l,
                voltage: u,
                threePhase: three,
                secim: next,
              ) ??
              double.nan);
      gerekce = text;
